rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
 
    function isLoggedIn() {
      return request.auth != null;
    }

   
    match /temperatures/{document} {
      allow read: if isLoggedIn();   
      allow write, update, delete: if false;      
   }
 }
}
